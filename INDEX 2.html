Step 6: Add Action 3 (Extract Jobs)

6 Extract Jobs
  1. Tap + to add action
  2. Search: Code by Zapier
  3. Select Python
  4. Delete default code, paste this:
import requests
import hashlib

url = input_data.get('url')
if not url:
    return {'error': 'No URL'}

sgai_key = "sgai-c3f2f2ef-47bb-4d72-b188-0729995d11a6"

prompt = """Extract every construction job listing visible.
Only Utah jobs. Only trades: roofing, framing, concrete, electrical, 
plumbing, hvac, excavation, landscaping, painting, flooring, general_labor, project_management.
For each job return: title, company, location, county, pay, job_type, description, posted_date, trade_category."""

payload = {
    "url": url,
    "prompt": prompt,
    "schema": {
        "jobs": {
            "type": "array",
            "items": {
                "type": "object",
                "properties": {
                    "title": {"type": ["string", "null"]},
                    "company": {"type": ["string", "null"]},
                    "location": {"type": ["string", "null"]},
                    "county": {"type": ["string", "null"]},
                    "pay": {"type": ["string", "null"]},
                    "job_type": {"type": ["string", "null"]},
                    "description": {"type": ["string", "null"]},
                    "posted_date": {"type": ["string", "null"]},
                    "trade_category": {"type": ["string", "null"]}
                }
            }
        }
    }
}

headers = {
    "Authorization": f"Bearer {sgai_key}",
    "Content-Type": "application/json"
}

try:
    response = requests.post(
        "https://api.scrapegraphai.com/v1/extract",
        json=payload,
        headers=headers,
        timeout=30
    )
    data = response.json()
    jobs = data.get('jobs', [])
    
    return {
        'jobs': jobs,
        'url': url,
        'count': len(jobs),
        'success': True
    }
except Exception as e:
    return {
        'error': str(e),
        'url': url,
        'success': False
    }

Test failed: Check internet connection. Try again — may be rate limit.
No jobs extracted: Site may require JavaScript. Check if URLs are correct.
Authorization failed: Check Bearer token has no typos. Make sure entire token is copied.
Zap turned off: Zapier may pause if errors persist. Turn it back on manually.
